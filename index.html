from openai import OpenAI

from config.settings import OPENAI_API_KEY


client = OpenAI(api_key=OPENAI_API_KEY)


def evaluate_content(content):
    response = client.responses.create(
        model="gpt-5.4-mini",
        input=f"""
Sen profesyonel bir YouTube Shorts editörüsün.

Aşağıdaki içeriği değerlendir.

İÇERİK:
{content}

Her kriteri 20 üzerinden puanla:

HOOK
AKICILIK
MERAK
BAŞLIK
SHORTS_UYGUNLUK

Toplam puan 100 üzerinden olsun.

Değerlendirmeyi şu formatta yap:

[PUAN]
Toplam puan: X/100

[DEGERLENDIRME]
HOOK: X/20 — kısa değerlendirme
AKICILIK: X/20 — kısa değerlendirme
MERAK: X/20 — kısa değerlendirme
BAŞLIK: X/20 — kısa değerlendirme
SHORTS_UYGUNLUK: X/20 — kısa değerlendirme

[SONUC]
ONAY veya RED

ÖNEMLİ:
Toplam puan 85 veya üzerindeyse SONUC ONAY olmalı.
Toplam puan 85'in altındaysa SONUC RED olmalı.

Sadece bu formatı kullan.
""",
    )

    editor_result = response.output_text

    print()
    print("=== AI EDITÖR ===")
    print(editor_result)

    # -----------------------------
    # OTOMATİK KARAR
    # -----------------------------

    decision = "RED"

    for line in editor_result.splitlines():

        if "Toplam puan:" in line:
            try:
                score_text = line.split(":", 1)[1]
                score = int(score_text.split("/")[0].strip())

                if score >= 85:
                    decision = "ONAY"
                else:
                    decision = "RED"

            except (ValueError, IndexError):
                decision = "RED"

    print()
    print("🎯 OTOMATİK KARAR:", decision)

    return editor_result, decision

def generate_images_from_prompt_file(prompt_file, output_folder):
    prompt_file = Path(prompt_file)
    output_folder = Path(output_folder)

    if not prompt_file.exists():
        raise FileNotFoundError(
            f"Görsel prompt dosyası bulunamadı: {prompt_file}"
        )

    output_folder.mkdir(parents=True, exist_ok=True)

    text = prompt_file.read_text(encoding="utf-8")

    prompts = [
        prompt.strip()
        for prompt in text.split("\n")
        if prompt.strip()
    ]

    if not prompts:
        raise ValueError("Görsel promptu bulunamadı.")

    print()
    print(f"🎨 Toplam {len(prompts)} görsel üretilecek.")

    generated_files = []

    for index, prompt in enumerate(prompts, start=1):

        print()
        print(f"🎨 Görsel {index}/{len(prompts)} oluşturuluyor...")

        output_path = output_folder / f"{index:02d}.png"

        generate_image(prompt, output_path)

        print(f"✅ Kaydedildi: {output_path}")

        generated_files.append(output_path)

    print()
    print("🎉 TÜM GÖRSELLER TAMAMLANDI!")

    return generated_files

def load_topics():

    if not TOPICS_FILE.exists():
        raise FileNotFoundError(
            f"topics.txt bulunamadı: {TOPICS_FILE}"
        )

    topics = []

    for line in TOPICS_FILE.read_text(
        encoding="utf-8"
    ).splitlines():

        topic = line.strip()

        if topic:
            topics.append(topic)

    if not topics:
        raise ValueError(
            "topics.txt boş. En az bir konu eklemelisin."
        )

    return topics


# ============================================================
# TOPLU ÜRETİM
# ============================================================

def run_batch():

    print()

    print(
        f"📚 Toplam konu: {len(topics)}"
    )

    print()

    successful = 0
    failed = 0

    for index, topic in enumerate(

        print()

Sen profesyonel bir YouTube Shorts içerik üreticisisin.

Konu: {topic}

Bu konu için bir Shorts içerik paketi hazırla.

Aşağıdaki bölümleri TAM OLARAK bu başlıklarla oluştur:

[SENARYO]
30-60 saniyelik, akıcı ve merak uyandırıcı Shorts senaryosu.
İlk cümle güçlü bir hook olsun.
Sonunda izleyiciye bir soru sor.

[BASLIK]
Merak uyandıran, tıklanabilir ama yanıltıcı olmayan bir YouTube Shorts başlığı.

[ACIKLAMA]
Kısa ve doğal bir YouTube açıklaması.

[HASHTAG]
Konuyla alakalı 5-8 hashtag.

[GÖRSEL_PROMPT]
Bu Shorts için kullanılabilecek 5 farklı görsel üretim promptu.
Her prompt ayrı satırda olsun.

Sadece bu bölümleri oluştur.
""",
    )

    content = response.output_text

    # ========================================================
    # 2. AI EDİTÖR KONTROLÜ
    # ========================================================

    print()

    editor_result, decision = evaluate_content(content)

        print("❌ İÇERİK REDDEDİLDİ")
        print("Dosya kaydedilmeyecek.")

        return

    print()

    # ========================================================
    # 3. KALİTE KONTROL
    # ========================================================

    required_sections = [
        "[SENARYO]",
        "[BASLIK]",
        "[ACIKLAMA]",
        "[HASHTAG]",
        "[GÖRSEL_PROMPT]",
    ]

    missing_sections = [
        section
        for section in required_sections
        if section not in content
    ]

    if missing_sections:

        print()
        print("⚠️ KALİTE KONTROL BAŞARISIZ")
        print("Eksik bölümler:")

        for section in missing_sections:
            print("-", section)

        print()
        print("Dosya kaydedilmedi.")

        return

    # ========================================================
    # 4. SHORTS KLASÖRÜ
    # ========================================================

    shorts_folder = Path("shorts")

    # ========================================================
    # 5. SIRADAKİ VİDEO NUMARASI

    existing_folders = [

        if folder.is_dir()

    if existing_folders:

def create_ai_captions(text):

    response = client.responses.create(
        model="gpt-5.4-mini",
        input=f"""
Aşağıdaki Türkçe konuşma metnini YouTube Shorts altyazıları için
doğal ve okunabilir kısa parçalara böl.

KURALLAR:

- Metni yaklaşık 14-18 anlamlı altyazı bloğuna böl.
- Her blok yaklaşık 4-9 kelime olabilir.
- Kelime sayısından çok anlam bütünlüğünü önemse.
- Doğal konuşma ritmini koru.
- Birbirine bağlı kelimeleri ayırma.
- "mi?", "mı?", "mu?", "mü?" gibi soru eklerini asla tek başına bırakma.
- "ve", "ama", "fakat", "ancak", "çünkü", "ile" gibi bağlaçları
  anlamsız şekilde ayırma.
- Çok kısa anlamsız parçalar oluşturma.
- Çok uzun parçalar oluşturma.
- Noktalama işaretlerini koru.
- Metindeki hiçbir kelimeyi değiştirme.
- Hiçbir kelimeyi silme.
- Yeni kelime ekleme.
- Kelimelerin sırasını değiştirme.

ÖNCELİK:

1. Anlam bütünlüğü
2. Doğal konuşma ritmi
3. Okunabilirlik
4. Yaklaşık 14-18 blok

SADECE JSON DÖNDÜR.

FORMAT:

{{
    "captions": [
        "parça 1",
        "parça 2",
        "parça 3"
    ]
}}

METİN:

{text}
""",
    )

    raw = response.output_text.strip()

    raw = re.sub(
        r"^```json\s*",
        "",
        raw,
        flags=re.IGNORECASE
    )

    raw = re.sub(
        r"\s*```$",
        "",
        raw
    )

    data = json.loads(raw)

    captions = data.get("captions", [])

    if not captions:
        raise ValueError(
            "AI altyazı parçaları oluşturamadı."
        )

    return captions

def get_audio_duration(audio_file):

    result = subprocess.run(
        [
            "ffprobe",
            "-v",
            "error",
            "-show_entries",
            "format=duration",
            "-of",
            "default=noprint_wrappers=1:nokey=1",
            str(audio_file),
        ],
        capture_output=True,
        text=True,
        check=True,
    )

    return float(result.stdout.strip())

def format_srt_time(seconds):

    milliseconds = int(round(seconds * 1000))

    hours = milliseconds // 3_600_000
    milliseconds %= 3_600_000

    minutes = milliseconds // 60_000
    milliseconds %= 60_000

    seconds = milliseconds // 1000
    milliseconds %= 1000

    return (
        f"{hours:02d}:"
        f"{minutes:02d}:"
        f"{seconds:02d},"
        f"{milliseconds:03d}"
    )

# ============================================================

def create_subtitle(
    transcript_file,
    audio_file,
    output_file,
):

    transcript_file = Path(transcript_file)
    audio_file = Path(audio_file)
    output_file = Path(output_file)

    if not transcript_file.exists():
        raise FileNotFoundError(
            f"Transkript bulunamadı: {transcript_file}"
        )

    if not audio_file.exists():
        raise FileNotFoundError(
            f"Ses dosyası bulunamadı: {audio_file}"
        )

    text = transcript_file.read_text(
        encoding="utf-8"
    ).strip()

    if not text:
        raise ValueError(
            "Transkript boş."
        )

    duration = get_audio_duration(
        audio_file
    )

    print()
    print("=== AI ALTYAZI MOTORU ===")
    print(
        f"🎙️ Ses süresi: {duration:.2f} saniye"
    )

    print(
        "🤖 AI doğal altyazı bloklarını oluşturuyor..."
    )

    captions = create_ai_captions(
        text
    )

    print(
        f"📝 {len(captions)} altyazı bloğu oluşturuldu."
    )

    # --------------------------------------------------------
    # KARAKTER TOPLAMI
    # --------------------------------------------------------

    total_characters = sum(
        len(caption)
        for caption in captions
    )

    if total_characters <= 0:
        raise ValueError(
            "Altyazı karakter sayısı geçersiz."
        )

    # --------------------------------------------------------
    # ZAMANLAMA
    # --------------------------------------------------------

    current_time = 0.0

    srt_blocks = []

    for index, caption in enumerate(
        captions,
        start=1
    ):

        ratio = (
            len(caption)
            / total_characters
        )

        segment_duration = (
            duration * ratio
        )

        start_time = current_time

        if index == len(captions):

            end_time = duration

        else:

            end_time = (
                current_time
                + segment_duration
            )

        srt_blocks.append(
            f"{index}\n"
            f"{format_srt_time(start_time)} --> "
            f"{format_srt_time(end_time)}\n"
            f"{caption}\n"
        )

        current_time = end_time

    # --------------------------------------------------------
    # KAYDET
    # --------------------------------------------------------

    output_file.parent.mkdir(
        parents=True,
        exist_ok=True
    )

    output_file.write_text(
        "\n".join(srt_blocks),
        encoding="utf-8"
    )

    print()
    print("=== ALTYAZI BLOKLARI ===")

    for index, caption in enumerate(
        captions,
        start=1
    ):
        print(
            f"{index:02d}: {caption}"
        )

    print()
    print("✅ AI ALTYAZI OLUŞTURULDU")
    print(f"Dosya: {output_file}")

    return output_file

import subprocess


# ============================================================
# SES SÜRESİ

def get_audio_duration(audio_file):

    result = subprocess.run(
        [
            "ffprobe",
            "-v",
            "error",
            "-show_entries",
            "format=duration",
            "-of",
            "default=noprint_wrappers=1:nokey=1",
            str(audio_file),
        ],
        capture_output=True,
        text=True,
        check=True,
    )

    return float(result.stdout.strip())


# ============================================================
# GÖRSELLER + SES → VİDEO
# ============================================================

def create_video(
    image_folder,
    audio_file,
    output_file,
):

    image_folder = Path(image_folder)
    audio_file = Path(audio_file)
    output_file = Path(output_file)

    # --------------------------------------------------------
    # DOSYA KONTROLLERİ
    # --------------------------------------------------------

    if not image_folder.exists():
        raise FileNotFoundError(
            f"Görsel klasörü bulunamadı: {image_folder}"
        )

    if not audio_file.exists():
        raise FileNotFoundError(
            f"Ses dosyası bulunamadı: {audio_file}"
        )

    images = sorted(
        image_folder.glob("*.png")
    )

    if not images:
        raise FileNotFoundError(
            f"Görsel bulunamadı: {image_folder}"
        )

    # --------------------------------------------------------
    # SES SÜRESİNİ ÖLÇ
    # --------------------------------------------------------

    audio_duration = get_audio_duration(
        audio_file
    )

    # --------------------------------------------------------
    # GÖRSELLERİN SÜRESİNİ HESAPLA
    # --------------------------------------------------------

    image_duration = (
        audio_duration / len(images)
    )

    print()
    print("=== VİDEO OLUŞTURMA ===")
    print(
        f"🎨 Görsel sayısı: {len(images)}"
    )
    print(
        f"🎙️ Ses süresi: {audio_duration:.2f} saniye"
    )
    print(
        f"⏱️ Görsel başına süre: "
        f"{image_duration:.2f} saniye"
    )
    print(
        "🎬 Video oluşturuluyor..."
    )

    # --------------------------------------------------------
    # GEÇİCİ GÖRSEL LİSTESİ
    # --------------------------------------------------------

    concat_file = (
        image_folder.parent
        / "image_list.txt"
    )

    with open(
        concat_file,
        "w",
        encoding="utf-8"
    ) as file:

        for image in images:

            file.write(
                f"file '{image.resolve()}'\n"
            )

            file.write(
                f"duration {image_duration:.6f}\n"
            )

        # FFmpeg concat demuxer'da son görselin
        # duration değerinin uygulanması için
        # son dosyayı tekrar ekliyoruz.

        file.write(
            f"file '{images[-1].resolve()}'\n"
        )

    # --------------------------------------------------------
    # FFMPEG
    # --------------------------------------------------------

    command = [
        "ffmpeg",
        "-y",

        "-f",
        "concat",

        "-safe",
        "0",

        "-i",
        str(concat_file),

        "-i",
        str(audio_file),

        # ----------------------------------------------------
        # SHORTS FORMAT
        # ----------------------------------------------------

        "-vf",
        (
            "scale=1080:1920:"
            "force_original_aspect_ratio=increase,"
            "crop=1080:1920"
        ),

        "-r",
        "25",

        # ----------------------------------------------------
        # VİDEO
        # ----------------------------------------------------

        "-c:v",
        "libx264",

        "-preset",
        "medium",

        "-crf",
        "20",

        "-pix_fmt",
        "yuv420p",

        # ----------------------------------------------------
        # SES
        # ----------------------------------------------------

        "-c:a",
        "aac",

        "-b:a",
        "128k",

        # ----------------------------------------------------
        # SESİ KESME
        # ----------------------------------------------------

        "-t",
        f"{audio_duration:.3f}",

        str(output_file),
    ]

    # --------------------------------------------------------
    # ÇALIŞTIR
    # --------------------------------------------------------

    subprocess.run(
        command,
        check=True
    )

    # --------------------------------------------------------
    # GEÇİCİ DOSYAYI SİL
    # --------------------------------------------------------

    if concat_file.exists():
        concat_file.unlink()

    # --------------------------------------------------------
    # SONUÇ
    # --------------------------------------------------------

    if not output_file.exists():
        raise RuntimeError(
            "FFmpeg çalıştı ancak video oluşmadı."
        )

    print()
    print("========================================")
    print("✅ VIDEO OLUŞTURULDU")
    print("========================================")
    print()
    print(
        f"Dosya: {output_file}"
    )

    return output_file

def generate_image(prompt, output_path):
    result = client.images.generate(
        model="gpt-image-2",
        prompt=prompt,
        size="1024x1024",
        quality="low",
    )

    image_data = result.data[0].b64_json

    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    with open(output_path, "wb") as f:
        f.write(base64.b64decode(image_data))

    return output_path
